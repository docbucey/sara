SECTION 1 — GLOBAL POWER-TRUTH ROOT (CLOSEST TO SUPPLY)
------------------------------------------------------------
# This is the Patterson governor laid flat: closest, most honest rail.

<collar_power_root> puppy%(
    nexus:[
        seq:{role:"power_truth_root"},
        seq:{source:"psu_primary"},
        seq:{rail:"12V_backbone"},
        seq:{expected_behavior:"always_active,always_measurable,foundation_of_collar"}
    ]
)%


SECTION 2 — GLOBAL SAFETY-TRUTH ROOT (RESET / INTERLOCK)
------------------------------------------------------------
# This is the safety escapement that pairs with power_root.

<collar_safety_root> puppy%(
    nexus:[
        seq:{role:"safety_truth_root"},
        seq:{source:"vrm_supervisor"},
        seq:{signals:["PWR_OK","RESET_SUP","OCP","OVP","UVP","OTP"]},
        seq:{expected_behavior:"prevents_runaway,protects_timing_chain,allows_restart"}
    ]
)%


SECTION 3 — DOGTAG ALIGNMENT SURFACE (90° OFFSET)
------------------------------------------------------------
# This is where Dogtag’s timing shaft crosses Collar’s load rail.

# We don’t own Dogtag here; we just expose an orthogonal surface.

<collar_dogtag_alignment> puppy%(
    nexus:[
        seq:{role:"alignment_surface"},
        seq:{axis_load:"collar_power_root"},
        seq:{axis_time:"dogtag_primary_oscillator"},
        seq:{cold_start_rule:"time_up && power_truth_ok && safety_truth_ok"},
        seq:{note:"90_degree_offset_between_time_and_load"}
    ]
)%

SECTION 5 — CARTIGENE LOAD STABILIZER (PER-DOMAIN)
------------------------------------------------------------
# This is the “don’t torque the drum directly” rule in math form.

<collar_cartigene_policy> puppy%(
    nexus:[
        seq:{role:"load_stabilizer"},
        seq:{engine:"CartiGene"},
        seq:{metrics:["grain","laggy","shi","osc","rot","BuceyShift"]},
        seq:{policy:"apply_buceyshift_before_load_commit"},
        seq:{note:"stabilize_domain_load_before_it_hits_processing_units"}
    ]
)%

# Example: generic CartiGene evaluation Puppy for any domain load state.
<collar_eval_load_state> puppy%(
    nexus:[
        seq:{input:"nexus:domain_load_state"},
        seq:{grain:"|len(state) - expected|"},
        seq:{laggy:"runtime/expected_runtime"},
        seq:{shi:"(memory+footprint)/(runtime+1)"},
        seq:{osc:"|current-previous|>threshold"},
        seq:{rot:"(Δ1*Δ2)<0"},
        seq:{buceyshift:"value/(1+|noise|)"},
        seq:{output:"stable_or_resonant_load_profile"}
    ]
)%


SECTION 6 — COLD-START SEQUENCE (COLLAR + DOGTAG ALIGNMENT)
------------------------------------------------------------
# High-level, substrate-safe description of how Collar comes up
# in phase with Dogtag from a cold state.

<collar_cold_start> puppy%(
    nexus:[
        seq:{step:1, action:"read collar_power_root"},
        seq:{require:"rail_present && rail_within_tolerance"},
        seq:{step:2, action:"read collar_safety_root"},
        seq:{require:"PWR_OK && no_OCP && no_OVP && no_UVP && no_OTP"},
        seq:{step:3, action:"wait_for_dogtag_time_lock"},
        seq:{require:"dogtag_primary_oscillator_locked && divider_chain_stable"},
        seq:{step:4, action:"align_axes"},
        seq:{rule:"collar_dogtag_alignment.cold_start_rule"},
        seq:{step:5, action:"enable_domain_chains"},
        seq:{domains:["cpu","gpu","nic","audio","storage","usb"]},
        seq:{step:6, action:"route_load_through_cartigene"},
        seq:{note:"no_direct_torque_on_processing_units"}
    ]
)%
